2. 伪代码的注释详细解释了代码的执行逻辑，当注释中有`【语义理解】`标记时，需要仔细阅读理解参数的内容和含义。

#伪代码#
```python
def get_action(new_info, old_memory):
    new_memory = []
    for info_item in new_info:
        info_save_flag = False
        for old_item in old_memory:
            if info_save_flag:
                continue
            if is_related_item(info_item, old_item):  # 【语义理解】判断提取信息对象info_item的fact内容和历史记忆对象old_item的fact内容是否有关联
                action = "UPDATE"
                memory_id = old_item.get("id")
                new_theme = merge_theme(info_item.get("theme"), old_item.get("theme"))
                new_time = select_latest_time(info_item.get("time"), old_item.get("time"))
                new_fact = []
                info_fact = info_item.get("fact")
                old_fact = old_item.get("fact")
                for info_content in info_fact:
                    rewrite_flag = False
                    for old_content in old_fact:
                        if rewrite_flag:
                            if exist_part_content_similar(info_content, old_content):  # 【语义理解】判断是否存在部分内容表达的含义相同
                                rewrite_flag = True
                                new_content = rewrite_content(info_content, old_content)
                                info_content = new_content
                        else:
                            new_fact.append(old_content)
                    if not rewrite_flag:
                        new_fact.append(info_content)
                new_memory.append({"id": memory_id, "theme": new_theme, "fact": new_fact, "time": new_time, "action": action})
                info_save_flag = True
                break
        if not info_save_flag:
            action = "ADD"
            new_memory.append({"id": get_unique_int_id(), "theme": info_item.get("theme"), "fact": info_item.get("fact"), "time": info_item.get("time"), "action": action})
    set_action_none_with_remaining_old_memory()
    check_unique_id(new_memory)
    return new_memory
```
